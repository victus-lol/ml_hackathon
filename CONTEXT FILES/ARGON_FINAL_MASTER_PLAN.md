    viterbi_arcs: list[int],
    arc_uv_map: dict[int, tuple[int, int]],
    node_pair_to_arc: dict[tuple[int, int], int],
    road_graph: "nx.DiGraph",
) -> tuple[list[int], float]:
    """
    1. Deduplicates consecutive identical arc_ids from Viterbi state sequence.
    2. Bridges any topological gaps (u_next != v_curr) via NetworkX shortest path.
    3. Computes exact Route Continuity score in [0.0, 1.0].
    """
    if not viterbi_arcs:
        return [], 1.0

    # Step 1: Contiguous run-length deduplication
    deduped: list[int] = [viterbi_arcs[0]]
    for arc_id in viterbi_arcs[1:]:
        if arc_id != deduped[-1]:
            deduped.append(arc_id)

    # Step 2: Dijkstra bridge splicing for skipped short intermediate arcs
    final_route: list[int] = [deduped[0]]
    for next_arc in deduped[1:]:
        prev_arc = final_route[-1]
        _, v_prev = arc_uv_map[prev_arc]
        u_next, _ = arc_uv_map[next_arc]

        if v_prev != u_next and road_graph.has_node(v_prev) and road_graph.has_node(u_next):
            try:
                node_path = nx.shortest_path(
                    road_graph, source=v_prev, target=u_next, weight="length_m"
                )
                # Splice intermediate bridge arcs (cap at 5 hops to prevent runaway loops)
                if 1 < len(node_path) <= 6:
                    for idx in range(len(node_path) - 1):
                        bridge_edge = (node_path[idx], node_path[idx + 1])
                        if bridge_edge in node_pair_to_arc:
                            bridge_arc = node_pair_to_arc[bridge_edge]
                            if bridge_arc != final_route[-1]:
                                final_route.append(bridge_arc)
            except nx.NetworkXNoPath:
                pass

        if next_arc != final_route[-1]:
            final_route.append(next_arc)

    # Step 3: Verify topological continuity across consecutive output arcs
    if len(final_route) <= 1:
        return final_route, 1.0

    connected_transitions = 0
    for i in range(len(final_route) - 1):
        _, v_curr = arc_uv_map[final_route[i]]
        u_nxt, _ = arc_uv_map[final_route[i + 1]]
        if v_curr == u_nxt:
            connected_transitions += 1

    continuity_ratio = connected_transitions / float(len(final_route) - 1)
    return final_route, round(continuity_ratio, 4)
